"""Experimental rank-one token scan with C64 checkpoints and a block-solve fallback."""

import torch
import triton

from fla.ops.key_gated_hyperbolic_delta_rule import sherman_morrison_kernels as kernels
from fla.ops.key_gated_hyperbolic_delta_rule.bf16_selective import (
    _require_runtime,
    _source_code_route,
    _validate_finite,
    chunk_metadata,
    launch,
)
from fla.ops.key_gated_hyperbolic_delta_rule.bf16_selective import (
    chunk_bf16 as block_chunk,
)
from fla.ops.key_gated_hyperbolic_delta_rule.fixed_curvature import curvature_tensor
from fla.ops.key_gated_hyperbolic_delta_rule.naive import _validate
from fla.ops.key_gated_hyperbolic_delta_rule.validation import FiniteValidationScope
from fla.utils import autocast_custom_bwd, autocast_custom_fwd, input_guard


class _ShermanMorrison(torch.autograd.Function):
    @staticmethod
    @input_guard
    @autocast_custom_fwd
    def forward(ctx, q, k, v, beta, ell, state, curve, opts, telemetry):
        h, kd, vd, gs = q.shape[2], q.shape[3], v.shape[-1], opts['g']
        ng, c, nseq = vd // gs, 64, state.shape[0]
        meta, count = chunk_metadata(q.device, tuple(opts['bounds']), c)
        saved = torch.empty((count*h*ng, kd, gs), dtype=torch.float32, device=q.device)
        output = torch.empty_like(v, dtype=q.dtype)
        final = torch.empty_like(state, dtype=torch.float32)
        risk = torch.empty((nseq*h*ng,), dtype=torch.int32, device=q.device)
        launch(kernels.sm_fwd_kernel, (nseq*h*ng,),
               (q, k, v, beta, ell, curve, state, output, final, *meta[:3], saved, risk,
                h, kd, vd, gs, ng, c, opts['scale']), telemetry, num_warps=4)
        ctx.save_for_backward(q, k, v, beta, ell, state, curve, *meta, saved)
        ctx.opts, ctx.telemetry = opts, telemetry
        ctx.set_materialize_grads(False)
        ctx.mark_non_differentiable(risk)
        if telemetry is not None:
            telemetry.update(sm_active=True, backend='triton_sm_token_scan', source_vjp='sm_rank_one_manual',
                             chunk_size=64, checkpoint_policy='canonical_entry_state_per_64_tokens',
                             token_parallel=False, state_control_precision='fp32', quality_control_precision='fp32',
                             contraction_precision='fp32_rank_one_with_bf16_or_fp32_sources',
                             sm_checkpoint_bytes=saved.numel()*saved.element_size(), host_fallback=False)
        return output, final, risk

    @staticmethod
    @input_guard
    @autocast_custom_bwd
    def backward(ctx, do, ds, risk_grad):
        if torch.is_grad_enabled():
            raise RuntimeError('first-order gradients only')
        saved, opts = ctx.saved_tensors, ctx.opts
        q, k, v, beta, ell, state, curve = saved[:7]
        meta, checkpoints = saved[7:12], saved[12]
        h, kd, vd, gs, c = q.shape[2], q.shape[3], v.shape[-1], opts['g'], 64
        ng, nseq, rows = vd//gs, state.shape[0], q.numel()//kd
        do = torch.zeros_like(v, dtype=q.dtype) if do is None else do.contiguous()
        ds = torch.zeros_like(state) if ds is None else ds.contiguous()
        grads = [torch.empty_like(x, dtype=torch.float32) for x in (q, k, v, beta, ell, state, curve)]
        replay = torch.empty((nseq*h*ng, c, kd, gs), dtype=torch.float32, device=q.device)
        partials = tuple(torch.empty((rows, ng, kd), dtype=torch.float32, device=q.device) for _ in range(3))
        partials += (torch.empty((rows, ng), dtype=torch.float32, device=q.device),)
        need_dc = ctx.needs_input_grad[6]
        dc = torch.empty((nseq*h*ng if need_dc else 0,), dtype=torch.float32, device=q.device)
        launch(kernels.sm_bwd_kernel, (nseq*h*ng,),
               (q, k, v, beta, ell, curve, do, ds, grads[5], grads[2], dc, *meta[:3], checkpoints, replay, partials,
                h, kd, vd, gs, ng, c, opts['scale'], need_dc), ctx.telemetry, num_warps=4)
        if rows:
            launch(kernels.sm_reduce_kernel, (triton.cdiv(rows, 16),),
                   (partials, grads[0], grads[1], grads[4], grads[3], rows, kd, ng, 16), ctx.telemetry, num_warps=4)
        grads[6] = dc.sum() if need_dc else torch.zeros_like(curve)
        result = [g.to(x.dtype) if need else None for g, x, need in
                  zip(grads, (q, k, v, beta, ell, state, curve), ctx.needs_input_grad[:7])]
        for group in opts['aliases']:
            owners = [i for i in group if ctx.needs_input_grad[i]]
            if owners:
                total = grads[group[0]]
                for i in group[1:]:
                    total = total + grads[i]
                for i in group:
                    result[i] = None
                result[owners[0]] = total.to(saved[owners[0]].dtype)
        _validate_finite(tuple(x for x in result if x is not None), opts['scope'],
                         'nonfinite SM gradients; FP64 training fallback forbidden')
        if ctx.telemetry is not None:
            ctx.telemetry.update(backward='sm_rank_one_manual', sm_backward_replay_bytes=replay.numel()*replay.element_size(),
                                 sm_source_partial_bytes=sum(x.numel()*x.element_size() for x in partials))
        return (*result, None, None)


def chunk_bf16(q, k, v, beta, log_decay, curvature=1., initial_state=None, output_final_state=False,
               scale=None, cu_seqlens=None, group_size=32, chunk_size=64, carry_mode='no_carry',
               arithmetic='bf16', telemetry=None, source_vjp='expanded', value_tile=32, state_value_tile=32,
               state_residual_mode='stable', validation_mode='strict', validation_scope=None):
    """Select the bounded C64 rank-one scan, or preserve the existing block solver.

    Internal state/control arithmetic is FP32; sources may be BF16 or FP32.
    Close quality margins, unqualified Key ranges and unsafe runtime normalization
    use the original block path. No FP64 training or hidden reference replay occurs.
    """
    _require_runtime(q)
    if chunk_size != 64 or group_size != 32 or q.shape[-1] != 128 or v.shape[-1] != 128:
        raise ValueError('SM research scan requires C64/K=V128/g32')
    if carry_mode != 'no_carry' or arithmetic != 'bf16' or state_residual_mode != 'stable':
        raise ValueError('SM research scan requires the stable no_carry path')
    if value_tile != 32 or state_value_tile != 32:
        raise ValueError('SM scan does not select the previous wide-tile candidates')
    if validation_mode not in ('strict', 'deferred'):
        raise ValueError('validation_mode must be strict or deferred')
    if validation_mode == 'deferred':
        if not torch.is_grad_enabled() or not isinstance(validation_scope, FiniteValidationScope) or not validation_scope.accepting:
            raise ValueError('SM deferred validation requires gradient-enabled execution and an open FiniteValidationScope')
    elif validation_scope is not None:
        raise ValueError('validation_scope requires deferred validation')
    bounds, scale = _validate(q, k, v, beta, log_decay, curvature, initial_state, scale, cu_seqlens, group_size, False)
    c = curvature_tensor(curvature, q.device)
    state = initial_state if initial_state is not None else torch.zeros(
        (len(bounds)-1, q.shape[2], q.shape[3], v.shape[-1]), dtype=torch.float32, device=q.device)
    sources = (q, k, v, beta, log_decay, state, c)
    if any(x.dtype not in (torch.float32, torch.bfloat16) for x in sources):
        raise ValueError('SM training accepts BF16/FP32 only')
    invalid, _ = _source_code_route(sources, group_size)
    if invalid:
        raise ValueError('invalid SM physical/source domain')
    aliases = {}
    for i, x in enumerate(sources):
        aliases.setdefault(id(x), []).append(i)
    options = dict(bounds=bounds, scale=scale, g=group_size, scope=validation_scope,
                   aliases=[g for g in aliases.values() if len(g) > 1])

    def fallback(reason):
        result = block_chunk(q, k, v, beta, log_decay, curvature=curvature, initial_state=initial_state,
                             output_final_state=output_final_state, scale=scale, cu_seqlens=cu_seqlens,
                             group_size=group_size, chunk_size=64, carry_mode=carry_mode, arithmetic=arithmetic,
                             source_vjp=source_vjp, validation_mode=validation_mode, validation_scope=validation_scope,
                             telemetry=telemetry)
        if telemetry is not None:
            telemetry.update(sm_requested=True, sm_active=False, sm_fallback=reason)
        return result

    output, final, risk = _ShermanMorrison.apply(*(x.contiguous() for x in sources), options, telemetry)
    if bool(risk.any()):
        return fallback('unsafe_fp32_quality_or_normalization')
    _validate_finite((output, final), validation_scope, 'nonfinite SM forward; FP64 training fallback forbidden')
    return output, final if output_final_state else None
